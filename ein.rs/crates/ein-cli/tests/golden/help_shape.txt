  OPTION --events -- metavar=FILE.jsonl arity=1 default=None choices=- group=- required=False help=record the engine's step-by-step event log to FILE (one JSON object per line). Off by default and additive: stdout, stderr and the exit code are unchanged. Schema: docs/kernel/inference/events.md.
  OPTION --events-level -- metavar=EVENTS_LEVEL arity=1 default='normal' choices=normal|verbose group=- required=False help=(--events) `verbose` adds redundant firings and pre-candidate hypothesis skips — ~6x the volume, and what T2 comparisons run at, since a dropped redundant firing is exactly what a port loses.
  OPTION --exhaustive -e metavar=- arity=0 default=False choices=- group=stop required=False help=do not stop at the first model. Certification also needs the lattice to end within --max-set-size, which `exhausted` in --stats reports (slower)

  OPTION --hyp-stats -H metavar=- arity=0 default=False choices=- group=- required=False help=print the root-hypothesis preview (total + per-relation breakdown + hypgen filter report)

  OPTION --lattice-order -o metavar=LATTICE_ORDER arity=1 default=None choices=lex|score-sum group=- required=False help=within-layer candidate ordering (S1.5b.26): 'lex' (canonical-tuple sort, the deterministic default) or 'score-sum' (S1.5a.7 hypgen scoring summed per set)
  OPTION --lattice-sanity-check -y metavar=- arity=0 default=False choices=- group=- required=False help=S1.5b.27 regression: for every alive size-k≥2 commitment, verify every (k-1)-subset parent path saturates to the same kb (costs k+1 saturations each)
  OPTION --layer-progress -- metavar=- arity=0 default=False choices=- group=- required=False help=stream the per-layer census to stderr and nothing per entering: three lines a layer — alive, then what the join proposed and the clauses filtered, then what entering them killed, learned and found

  OPTION --max-set-size -m metavar=MAX_SET_SIZE arity=1 default='5' choices=- group=- required=False help=commitment-set depth cap (default: 5)
  OPTION --max-time -T metavar=MAX_TIME arity=1 default=None choices=- group=- required=False help=abort after N wall-clock seconds
  OPTION --models -- metavar=FORM arity=1 default='list' choices=list|key group=- required=False help=how to print a model SET: 'list' (a block per model, the default) or 'key' (the smallest set of slots that tells them apart, and the table of combinations that occur). Read by an ambiguous verdict only; changes stdout and nothing recorded
  OPTION --no-diagrams -G metavar=- arity=0 default=False choices=- group=- required=False help=(--trace) suppress all inline dot blocks
  OPTION --no-kill-cache -K metavar=- arity=0 default=False choices=- group=- required=False help=disable the lookahead-kill (not h) cache (pairs with --no-lookahead to exercise the raw loop)
  OPTION --no-lookahead -L metavar=- arity=0 default=False choices=- group=- required=False help=disable hypgen's one-step lookahead (forces deaths through the no-good learning path)
  OPTION --print-final-hfacts -f metavar=- arity=0 default=False choices=- group=- required=False help=dump only the hypothesis-target (query :hrules) facts per solution
  OPTION --print-final-positive -P metavar=- arity=0 default=False choices=- group=- required=False help=like --print-final-state but drops the (not …) facts
  OPTION --print-final-state -p metavar=- arity=0 default=False choices=- group=- required=False help=dump each solution's derived facts (for an unsat verdict, the unsat-core facts instead)
  OPTION --progress-every -g metavar=PROGRESS_EVERY arity=1 default='100' choices=- group=- required=False help=under --verbose, log every N-th entering (default 100; set 1 to log every entering)
  OPTION --relevant -l metavar=- arity=0 default=False choices=- group=- required=False help=(--trace) prune to the goal-relevant slice
  OPTION --reorder -R metavar=- arity=0 default=False choices=- group=- required=False help=(--trace) cluster steps by target entity instead of engine order

  OPTION --stats -s metavar=- arity=0 default=False choices=- group=- required=False help=print engine counters (k, enterings, layers, wall)
  OPTION --timing -t metavar=- arity=0 default=False choices=- group=- required=False help=print a per-phase wall-clock table: parse, kb load, compile, root saturation, hypothesis search (enterings, layers, saturations, per-hypothesis avg), and totals

  OPTION --verbose -v metavar=- arity=0 default=False choices=- group=- required=False help=stream per-layer + per-entering progress to stderr
